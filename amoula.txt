def connect_to_server():
    try:
        # Informations d'authentification pour votre serveur SFTP
        host = '192.168.80.244'
        port = 300  # Port SFTP par défaut
        username = 'ftp-switch'
        password = '1234'

        # Établir une connexion SFTP
        transport = paramiko.Transport((host, port))
        transport.connect(username=username, password=password)
        sftp = paramiko.SFTPClient.from_transport(transport)

        # Récupérer la liste des fichiers et répertoires
        liste_fichiers_repertoires = sftp.listdir('.')
        
        # Fermer la connexion SFTP

        return liste_fichiers_repertoires

    except Exception as e:
        print(f"Une erreur s'est produite : {str(e)}")
        return None
